#分配uid

execute as @r[scores={钓鱼UID=0}] at @s run scoreboard players operation @s 钓鱼UID = 钓鱼UID 钓鱼UID


#开始检测
scoreboard players add @e[type=fishing_hook] 钓鱼UID 0

execute as @e[type=fishing_hook,scores={钓鱼UID=0}] at @s at @p[r=1.5] run scoreboard players operation @s 钓鱼UID = @p[hasitem={item=tropical_fish,location=slot.hotbar,slot=0,quantity=1..}] 钓鱼UID

#检测鱼钩进水
execute as @e[type=fishing_hook,scores={钓鱼UID=1..},tag=!入水] at @s if block ~~0.1~ water run tag @s add 入水

#检测入水后浮出
execute as @e[type=fishing_hook,scores={钓鱼UID=1..},tag=入水] at @s unless block ~~0.1~ water run tag @s add 浮出

#检测浮出后入水
execute as @e[type=fishing_hook,scores={钓鱼UID=1..},tag=浮出] at @s if block ~~0.1~ water run tag @s add 咬钩

#如果咬钩后没有钓上来检测未咬钩
execute as @e[type=fishing_hook,scores={钓鱼UID=1..},tag=咬钩] at @s unless block ~~0.1~ water run tag @s add 未咬钩

#如果未咬钩检测到水就是咬钩了
execute as @e[type=fishing_hook,scores={钓鱼UID=1..},tag=未咬钩] at @s if block ~~0.1~ water run tag @s add 咬钩

execute as @e[type=fishing_hook,tag=咬钩] at @s if block ~~0.1~ water run tag @s remove 未咬钩

execute as @e[type=fishing_hook,tag=未咬钩] at @s unless block ~~0.1~ water run tag @s remove 咬钩

#检测咬钩后给上钩tag
execute as @e[type=fishing_hook,tag=咬钩] at @s at @a[rm=0.1] if score @s 钓鱼UID = @p 钓鱼UID run tag @p add 上钩了

execute as @e[type=fishing_hook,tag=咬钩,tag=!鱼钩咬钩] at @s at @a[rm=0.1] if score @s 钓鱼UID = @p 钓鱼UID run tag @p[tag=上钩了,tag=!获得物资,hasitem={item=nautilus_shell,data=520,quantity=1..,location=slot.weapon.offhand}] add 获得物资

execute as @e[type=fishing_hook,tag=咬钩] at @s run tag @s add 鱼钩咬钩

execute as @e[type=fishing_hook,tag=未咬钩] at @s run tag @s remove 鱼钩咬钩

execute as @a[tag=上钩了] at @s at @e[rm=0.1] if score @s 钓鱼UID = @e[c=1,type=fishing_hook] 钓鱼UID run tag @s add 鱼钩还存在

execute as @a[tag=上钩了,tag=!鱼钩还存在] at @s run tag @s add 获得物资

execute as @e[type=fishing_hook,tag=未咬钩] at @s at @a[rm=0.1] if score @s 钓鱼UID = @p 钓鱼UID run tag @p remove 上钩了

tag @a[tag=获得物资] remove 上钩了

tag @a remove 鱼钩还存在
